advent_of_code::solution!(%DAY_NUMBER%);

type Answer = usize;

#[derive(PartialEq)]
enum Part {
    One = 1,
    Two = 2,
    Three = 3,
}

fn solve(input: &str, part: Part, debug: bool) -> Option<Answer> {
    match part {
        Part::One => None,
        Part::Two => None,
        Part::Three => None,
    }
}

pub fn part_one(input: &str) -> Option<Answer> {
    solve(input, Part::One, false)
}

pub fn part_two(input: &str) -> Option<Answer> {
    solve(input, Part::Two, false)
}

pub fn part_three(input: &str) -> Option<Answer> {
    solve(input, Part::Three, false)
}

        assert_eq!(result, None);

    #[test]
    fn test_part_two() {
        let input = advent_of_code::template::read_file_part("examples", DAY, 2);
        let result = solve(&input, Part::Two, true);
        assert_eq!(result, None);
    }

    #[test]
    fn test_part_three() {
        let input = advent_of_code::template::read_file_part("examples", DAY, 3);
        let result = solve(&input, Part::Three, true);
        assert_eq!(result, None);
    }
}
